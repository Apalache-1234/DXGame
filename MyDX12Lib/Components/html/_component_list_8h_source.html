319 bool IsParentComponent(UINT32 compHash , UINT32 treeIndex , UINT32 leafIndex)
320 {
321 const UINT32& parentHash = COMPONENT_HASH<T>;
322 LeafData childLeaf = tree[treeIndex][leafIndex];
323 int depth = treeIndex;
324
325 int index = childLeaf.parentIndex;
326
327 while (index != -1 || depth != 0)
328 {
329 depth--;
330 if (tree[depth][index].hash == parentHash)
331 return true;
332 index = tree[depth][index].parentIndex;
333 }

335 return false;
